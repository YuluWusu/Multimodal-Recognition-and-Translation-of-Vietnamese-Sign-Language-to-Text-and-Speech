# 3. ĐỊNH NGHĨA DATASET (ĐÃ SỬA LỖI allow_pickle & PADDING)
class VSL400KeypointDataset(Dataset):
    def __init__(self, npz_path, max_len=60):
        # Mở file cho phép allow_pickle=True
        data = np.load(npz_path, allow_pickle=True)
        self.X = data['X']
        self.y = np.array(data['y'], dtype=np.int64)
        self.max_len = max_len

    def __len__(self):
        return len(self.y)

    def __getitem__(self, idx):
        # Ép từng sample về kiểu numpy float32
        x_sample = np.array(self.X[idx], dtype=np.float32)
        
        # Nếu dữ liệu dạng 2D (seq_len, feature_dim), ta tiến hành Padding/Truncate
        if x_sample.ndim == 2:
            seq_len, feature_dim = x_sample.shape
            if seq_len >= self.max_len:
                # Cắt bớt nếu dài hơn MAX_SEQ_LEN
                x_padded = x_sample[:self.max_len, :]
            else:
                # Chèn số 0 (padding) nếu ngắn hơn MAX_SEQ_LEN
                x_padded = np.zeros((self.max_len, feature_dim), dtype=np.float32)
                x_padded[:seq_len, :] = x_sample
            x_sample = x_padded

        return torch.from_numpy(x_sample), torch.tensor(self.y[idx], dtype=torch.long)

# 4. NAP DATALOADER FOLD 1
